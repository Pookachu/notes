
$$\begin{aligned}
&\lim_{x\to\infty} \frac{\sqrt{49x^6+5-1}}{7x^3-2} \\\\
=& \lim_{x\to\infty} \frac{\sqrt{49x^6(1+\frac{5x}{49x^6}-\frac{1}{49x^6})}}{7x^3-2} \\\\
=& \lim_{x\to\infty} \frac{\sqrt{49x^6}\sqrt{1+\frac{5x}{49x^6}-\frac{1}{49x^6}}}{7x^3-2} \\\\
=& \lim_{x\to\infty} \frac{\sqrt{49}\sqrt{x^6}\sqrt{1+\frac{5x}{49x^6}-\frac{1}{49x^6}}}{7x^3-2} \\\\
=& \lim_{x\to\infty} \frac{7x^3\sqrt{1+\frac{5x}{49x^6}-\frac{1}{49x^6}}}{7x^3-2} \\\\
=& \lim_{x\to\infty} \frac{7x^3\sqrt{1+\frac{5x}{49x^6}-\frac{1}{49x^6}}}{7x^3(1-\frac{2}{7x^3})} \\\\
=& \lim_{x\to\infty} \frac{7\sqrt{1+\frac{5x}{49x^6}-\frac{1}{49x^6}}}{7(1-\frac{2}{7x^3})} \\\\
=&\quad \frac{7(1+0-0)}{7(1-0)} \quad=\quad \frac{7}{7} \quad=\quad \boxed{1}
\end{aligned}$$
